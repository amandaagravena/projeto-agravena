criação da função def_pol

def_pol <- function(x, y, pol){
  as.logical(sp::point.in.polygon(point.x = x,
                                  point.y = y,
                                  pol.x = pol[,1],
                                  pol.y = pol[,2]))
}

library(sf)
library(dplyr)
library(ggplot2)
library(rnaturalearth)   # precisa também do pacote rnaturalearthdata

sf_use_s2(FALSE)

crs_m   <- 5880   # SIRGAS 2000 / Brazil Polyconic (metros)
dist_km <- 50     # largura da faixa costeira

# 1) Brasil continental (sem ilhas), em metros
br <- brasil_cont |>
  st_union() |>
  st_make_valid() |>
  st_transform(crs_m)

# 2) Países vizinhos (inclui a Guiana Francesa, que no Natural Earth está em "France")
vizinhos <- ne_countries(scale = "medium", returnclass = "sf") |>
  filter(admin != "Brazil") |>
  st_crop(xmin = -85, xmax = -30, ymin = -60, ymax = 15) |>
  st_union() |>
  st_make_valid() |>
  st_transform(crs_m)

# 3) Linha de costa = contorno do Brasil menos as fronteiras terrestres
#    (tolerância de 10 km porque os limites do geobr e do Natural Earth não coincidem exatamente)
costa <- st_boundary(br) |>
  st_difference(st_buffer(vizinhos, 10000))

# 4) Faixa costeira só do lado continental
zona_costeira <- costa |>
  st_buffer(dist_km * 1000) |>
  st_intersection(br) |>
  st_make_valid() |>
  st_transform(4326) |>
  st_sf()

# Conferência visual
ggplot() +
  geom_sf(data = brasil_cont, fill = "grey95", color = "grey50", linewidth = 0.3) +
  geom_sf(data = zona_costeira, fill = "steelblue", alpha = 0.6, color = NA)


# 5) Flag dos pontos da malha
pts <- st_as_sf(grid_am_cut, coords = c("lon", "lat"), crs = 4326, remove = FALSE)
grid_am_cut$flag_am <- lengths(st_intersects(pts, zona_costeira)) > 0
