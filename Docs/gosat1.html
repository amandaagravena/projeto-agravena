Retirada de tendência (detrend)

a_ch4 <- mod_trend_xch4$coefficients[[1]]
b_ch4 <- mod_trend_xch4$coefficients[[2]]

df_gosat1_costeiro_buffer <- df_gosat1_costeiro_buffer |>
  mutate(
    date_modif = as.numeric(date - min(date)),
    xch4_est = a_ch4 + b_ch4 * date_modif,
    delta = xch4_est - xch4,
    xch4_detrend = (a_ch4 - delta) - (mean(xch4, na.rm = TRUE) - a_ch4)
  )

write_rds(df_gosat1_costeiro_buffer, "../data/xch4-gosat1-costeiro-terrestre-buffer.rds")

Criar as faixas de latitude e aplicar os 5 grupos definidos pelo XCO₂

## Criar as faixas de latitude e aplicar os 5 grupos definidos pelo XCO₂

df_gosat1_grupos <- df_gosat1_costeiro_buffer |>
  filter(year > 2014) |> 
  mutate(
    class_latitude = cut(latitude, 115),
    year_month = paste(
      as.character(year),
      as.character(month),
      sep = "_"
    ),
    latitude_media = (
      as.numeric(sub("\\(([-0-9.]+),.*", "\\1", class_latitude)) +
      as.numeric(sub(".*[,]([-0-9.]+)\\]", "\\1", class_latitude))
    ) / 2,
    grupo = case_when(
      latitude <= -28.6 ~ 5,
      latitude > -28.6 & latitude <= -22.5 ~ 4,
      latitude > -22.5 & latitude <= -10.5 ~ 3,
      latitude > -10.5 & latitude <= 0.37 ~ 2,
      latitude > 0.37 ~ 1
    )
  )

df_gosat1_grupos |>
  count(grupo) |>
  arrange(grupo)
