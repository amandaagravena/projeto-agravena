    coords = c("longitude", "latitude"),
    crs = 4326,
    remove = FALSE
  )

biomes <- sf::st_transform(biomes, crs = 4326)

  costeiro_terrestre_buffer,

  left = TRUE

## Retirada de tendência (detrend)

a_ch4_gosat2 <- mod_trend_xch4_gosat2$coefficients[[1]]
b_ch4_gosat2 <- mod_trend_xch4_gosat2$coefficients[[2]]

df_gosat2_costeiro_buffer <- df_gosat2_costeiro_buffer |>
  mutate(
    date_modif = as.numeric(date - min(date)),
    xch4_est = a_ch4_gosat2 + b_ch4_gosat2 * date_modif,
    delta = xch4_est - xch4,
    xch4_detrend = (
      a_ch4_gosat2 - delta
    ) - (
      mean(xch4, na.rm = TRUE) - a_ch4_gosat2
    )
  )

write_rds(
  df_gosat2_costeiro_buffer,
  "../data/xch4-gosat2-costeiro-terrestre-buffer.rds"
)
df_gosat2_grupos <- df_gosat2_costeiro_buffer |>
  filter(year > 2014) |> 
  mutate(
    class_latitude = cut(latitude, 115),
    year_month = paste(
      as.character(year),
      as.character(month),
      sep = "_"
    ),
    latitude_media = (
      as.numeric(sub(
        "\\(([-0-9.]+),.*",
        "\\1",
        class_latitude
      )) +
      as.numeric(sub(
        ".*[,]([-0-9.]+)\\]",
        "\\1",
        class_latitude
      ))
    ) / 2,
    grupo = case_when(
      latitude <= -28.6 ~ 5,
      latitude > -28.6 & latitude <= -22.5 ~ 4,
      latitude > -22.5 & latitude <= -10.5 ~ 3,
      latitude > -10.5 & latitude <= 0.37 ~ 2,
      latitude > 0.37 ~ 1
    )
  )
